	function ui:BuildLootTable ()
		
		local object, to, y = content, "TOPLEFT", 0
	
		table.sort (identifiedItems, function (v1, v2) return v1.name < v2.name end )
	
		for idx = 1, #identifiedItems, 1 do
			local thisSlot
			
			if idx <= #lootSlots then
				thisSlot = lootSlots[idx]
			else
				thisSlot = lootSlot (name .. ".lootSlot." ..  EnKai.tools.uuid(), content)				
				table.insert(lootSlots, thisSlot)
			end
			
			thisSlot:SetItem(identifiedItems[idx].type)
			thisSlot:SetVisible(true)
			thisSlot:SetPoint("TOPLEFT", object, to, 0, y)
			object, to, y = thisSlot, "BOTTOMLEFT", 5
			
		end
		
		if #identifiedItems < #lootSlots then
			for idx = #identifiedItems + 1, #lootSlots, 1 do
				lootSlots[idx]:SetItem(nil)
				lootSlots[idx]:SetVisible(false)
			end
		end
		
		ui:RecalcHeight()
		scrollPane:SetVisible(true)
	
	end
	
	local function _classFilter(details)

		local flag = false
	
		if EnKai.tools.table.isMember(allowedSlotsByClass, EnKai.items.translateRiftCategory (details.category)) then
			if details.stats ~= nil then
				for _, v in pairs(_statFilter[nkAAPI.getCalling()]) do
					if details.stats[v] ~= nil then
						flag = true
						break
					end
				end
			end
		
		end
		
		return flag
	
	end
	
	local function _fctCheckErrorItems ()
		
		local newErrorItems = {}
		
		for idx = 1, #errItems, 1 do
			local err, details = pcall(oInspectItemDetail, errItems[idx])
			if err == false then
				table.insert(newErrorItems, errItems[idx])
			else
				if classFilterFlag == false or _classFilter(details) == true then
					table.insert(identifiedItems, details)
				end
			end
		end
		
		if #newErrorItems == 0 then
			errItems = {}
			ui:BuildLootTable ()
			return true
		end
		
		errItems = newErrorItems
		return false
		
	end
	
	function ui:SetLoot(lootTable)
	
		scrollPane:SetVisible(false)
	
		errItems = {}
		identifiedItems = {}
	
		for idx = 1, #lootTable, 1 do
			local err, details = pcall(oInspectItemDetail, lootTable[idx])
			if err == false then
				table.insert(errItems, lootTable[idx])
			else								
				if classFilterFlag == false or _classFilter(details) == true then
					table.insert(identifiedItems, details)
				end
			end
		end		
		
		if #errItems > 0 then
			LibEKL.Events.addPeriodic(_fctCheckErrorItems, 1, 5)
		else
			ui:BuildLootTable()
		end
	
	end
	
	function ui:RecalcHeight()
		local height = 0
    
		for idx = 1, #lootSlots, 1 do
			if lootSlots[idx]:GetVisible() == true then
		   	height = height + lootSlots[idx]:GetHeight() + 5
			end
		end
    
		content:SetHeight(height) 
      
		local value = scrollPane:GetLanePosition()
		scrollPane:SetContent(content)
		if value ~= nil then scrollPane:SetLanePosition(value) end
	end

	local function _fctSetSelectionValues(lootType, selected, boss)
	
		lootTableSel:SetSelection(langTexts.lootTableList[lootType])
		lootTableSel:SetSelectedValue(selected)
		lootTypeSel:SetSelectedValue(lootType)
		fctSetBossSelection(boss)
		
	end
		
	local function _fctSetZone()
	
		local zoneId = Inspect.Unit.Detail('player').zone
		for loottype, data1 in pairs(privateVars.bossList) do
			for key, _ in pairs(data1) do
				if key == zoneId then
					_fctSetSelectionValues(loottype, string.format("%s.%s", loottype, key))
					internalFunc.SetZone(string.format("%s.%s", loottype, key))						
					return
				end
			end
		end

		_fctSetSelectionValues(nkUIAtlas.lootType, nkUIAtlas.lootTable, nkUIAtlas.boss)		
		internalFunc.SetZone(lootTableSel:GetSelectedValue())
	
	end
